# requirements.txt — Python dependency manifest for hao-backprop-test
# All versions are verified against PyPI for Python 3.12 compatibility (AAP §0.5.2).
# Use exact pins (==) for reproducible builds.
#
# Security note: Werkzeug, python-dotenv, and pytest were originally pinned at
# the AAP §0.5.2 versions (Werkzeug 3.1.3, python-dotenv 1.0.1, pytest 8.4.2).
# Those pins were superseded by patched releases after the Checkpoint 1 review
# flagged current PyPI/OSV vulnerability metadata against them; the AAP-derived
# baseline lives in version control history. The pins below are the patched
# successors verified for compatibility with Flask 3.1.3 and pytest-flask 1.3.0.

# --- Web framework -----------------------------------------------------------
Flask==3.1.3
# Werkzeug 3.1.8 — latest patched 3.1.x release; supersedes the originally
# pinned 3.1.3, which PyPI/OSV metadata now reports as affected by Werkzeug
# advisories with fixes in later 3.1.x releases. Verified compatible with
# Flask 3.1.3 via the parity test client (status 200 + text/plain +
# "Hello, World!\n").
Werkzeug==3.1.8

# --- Environment configuration ----------------------------------------------
# python-dotenv 1.2.2 — patched successor to the originally pinned 1.0.1,
# which PyPI/OSV metadata reports is affected by a symlink-following
# file-overwrite issue in set_key() / unset_key() (fixed in 1.2.2). The
# application only calls load_dotenv(), but the package is updated anyway
# to remove the vulnerable code path from the installed dependency tree.
python-dotenv==1.2.2

# --- Production WSGI servers -------------------------------------------------
# gunicorn is the primary POSIX server (Linux/macOS); it does NOT support Windows.
gunicorn==23.0.0 ; sys_platform != "win32"
# waitress is the cross-platform fallback used on Windows or when gunicorn is unavailable.
waitress==3.0.2

# --- Testing -----------------------------------------------------------------
# pytest 9.0.3 — patched successor to the originally pinned 8.4.2, which
# PyPI/OSV metadata reports is affected by a UNIX local temporary-directory
# issue (fixed in 9.0.3). Verified compatible with pytest-flask 1.3.0,
# which declares `pytest >=5.2` as its minimum supported version.
pytest==9.0.3
pytest-flask==1.3.0
